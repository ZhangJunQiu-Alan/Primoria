# Simulates electromagnetic wave propagation
import numpy as np

def wave(x, t, frequency=5e14):
    """Returns E-field amplitude at position x and time t."""
    wavelength = 3e8 / frequency  # c = f × λ
    k = 2 * np.pi / wavelength
    omega = 2 * np.pi * frequency
    return np.cos(k * x - omega * t)

x = np.linspace(0, 1e-6, 1000)
E = wave(x, t=1e-15)
